3.KUBERNETS

:: Start Docker
docker --version
docker ps
docker pull nginx

:: Start Minikube
minikube start --driver=docker
minikube status
kubectl get nodes

:: Create deployment with 2 replicas
kubectl create deployment ganesh --image=nginx --replicas=2
kubectl get deployments
kubectl get pods

:: Expose deployment
kubectl expose deployment ganesh --type=NodePort --port=80 --target-port=80
kubectl get services

:: Access application (keep running in separate terminal)
kubectl port-forward service/ganesh 8899:80
:: Open http://localhost:8899
